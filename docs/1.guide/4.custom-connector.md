# Custom Connectors

> Create your own connector for databases not yet supported by db0.

## Connector Interface

A connector must implement the `Connector` interface:

```ts
interface Connector<TInstance = unknown> {
  name: string;
  dialect: "mysql" | "postgresql" | "sqlite" | "libsql" | "mssql";
  capabilities: DatabaseCapabilities;
  getInstance: () => TInstance | Promise<TInstance>;
  exec: (sql: string) => ExecResult | Promise<ExecResult>;
  prepare: (sql: string) => Statement;
  dispose?: () => void | Promise<void>;
  beginTransaction?: (opts?: TransactionOptions) => Promise<ConnectorTransaction>;
}
```

### Required Properties

| Property | Type | Description |
|----------|------|-------------|
| `name` | `string` | Unique connector name |
| `dialect` | `SQLDialect` | SQL dialect: `mysql`, `postgresql`, `sqlite`, or `libsql` |
| `capabilities` | `DatabaseCapabilities` | Feature flags indicating what the database supports |
| `getInstance` | `() => TInstance` | Returns the underlying client instance |
| `exec` | `(sql: string) => ExecResult` | Executes raw SQL |
| `prepare` | `(sql: string) => Statement` | Creates a prepared statement |

### Optional Properties

| Property | Type | Description |
|----------|------|-------------|
| `dispose` | `() => void` | Cleanup function called when database is closed |
| `beginTransaction` | `(opts?) => Promise<ConnectorTransaction>` | Native transaction support |

## Capabilities

Define what features your database supports:

```ts
const capabilities: DatabaseCapabilities = {
  supportsJSON: true,       // Native JSON columns/operations
  supportsBooleans: false,  // Native boolean type (vs integer 0/1)
  supportsArrays: false,    // Array columns
  supportsDates: false,     // Native date/timestamp columns
  supportsUUIDs: false,     // Native UUID type
  supportsTransactions: true, // Transaction support
  supportsBatch: true,      // Batch operations
};
```

## Statement Interface

The `prepare` method must return a `Statement`:

```ts
interface Statement {
  bind(...params: Primitive[]): PreparedStatement;
  all(...params: Primitive[]): Promise<unknown[]>;
  run(...params: Primitive[]): Promise<{ success: boolean }>;
  get(...params: Primitive[]): Promise<unknown>;
}
```

## Example: Custom Connector

```ts
import type { Connector, Primitive } from "db0";
import { BoundableStatement } from "db0/connectors/_internal/statement";

interface ConnectorOptions {
  connectionString: string;
}

export default function myConnector(opts: ConnectorOptions): Connector<MyClient> {
  let client: MyClient | undefined;

  const getClient = () => {
    if (!client) {
      client = new MyClient(opts.connectionString);
    }
    return client;
  };

  return {
    name: "my-database",
    dialect: "postgresql",
    capabilities: {
      supportsJSON: true,
      supportsBooleans: true,
      supportsArrays: false,
      supportsDates: true,
      supportsUUIDs: true,
      supportsTransactions: true,
      supportsBatch: false,
    },
    getInstance: () => getClient(),
    exec: (sql) => getClient().query(sql),
    prepare: (sql) => new MyStatement(sql, getClient()),
    dispose: () => {
      client?.close();
      client = undefined;
    },
  };
}

class MyStatement extends BoundableStatement<void> {
  #sql: string;
  #client: MyClient;

  constructor(sql: string, client: MyClient) {
    super();
    this.#sql = sql;
    this.#client = client;
  }

  async all(...params: Primitive[]) {
    const result = await this.#client.query(this.#sql, params);
    return result.rows;
  }

  async run(...params: Primitive[]) {
    const result = await this.#client.query(this.#sql, params);
    return { success: true, changes: result.rowCount };
  }

  async get(...params: Primitive[]) {
    const result = await this.#client.query(this.#sql, params);
    return result.rows[0];
  }
}
```

## Transaction Support

For native transaction support, implement `beginTransaction`:

```ts
beginTransaction: async (opts?: TransactionOptions): Promise<ConnectorTransaction> => {
  const client = getClient();
  await client.query("BEGIN");

  return {
    exec: (sql) => client.query(sql),
    prepare: (sql) => new MyStatement(sql, client),
    commit: () => client.query("COMMIT"),
    rollback: () => client.query("ROLLBACK"),
  };
};
```

## Using Your Connector

```ts
import { createDatabase } from "db0";
import myConnector from "./my-connector";

const db = createDatabase(
  myConnector({
    connectionString: "my://localhost/db",
  }),
);

const { rows } = await db.sql`SELECT * FROM users`;
```
